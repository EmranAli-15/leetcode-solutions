/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    bool hasCycle(ListNode *head) {
        ListNode* slow = head;
        ListNode* fast = head;

        int flag = 0;

        while(fast && fast->next)
        {
            if(!flag)
            {
                slow = slow->next;
                fast = fast->next->next;

                if(slow == fast)
                {
                    flag = 1;
                    slow = head;
                }
            }

            if(flag)
            {
                slow = slow->next;
                fast = fast->next;

                if(slow == fast) return true;
            }
        }
        return false;
    }
};
