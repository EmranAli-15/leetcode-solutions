class Solution {
public:
    int firstStableIndex(vector<int>& nums, int k) {
        
        int mx = nums[0];
        int mn;
        int idx = -1;

        for(int i=0; i<nums.size(); i++)
        {
            if(nums[i] > mx) mx=nums[i];
            mn = nums[i];
            
            for(int j=i; j<nums.size(); j++)
            {
                if(nums[j] < mn) mn=nums[j];
            }

            if(mx-mn <= k)
            {
                idx=i;
                break;
            }

        }

        return idx;
    }
};
