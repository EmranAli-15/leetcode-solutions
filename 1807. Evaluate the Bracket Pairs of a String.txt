class Solution {
public:
    string evaluate(string s, vector<vector<string>>& knowledge) {
        unordered_map<string, string> store;

        int len = knowledge.size();
        for(int i=0; i<len; i++)
        {
            string k = knowledge[i][0];
            string v = knowledge[i][1];
            store.insert({k, v});
        }

        string ans = "";
        int flag = 0;
        string search = "";
        for(auto c:s)
        {
            if(c != '(' && !flag)ans+=c;
            else{
                flag = 1;
                if(c == '(')continue;
                else if(c != ')')search+=c;
                else{
                    if(store.count(search)){
                        ans+=store.at(search);
                        flag = 0;
                        search = "";
                    }
                    else {
                        ans+="?";
                        search = "";
                        flag = 0;
                    }
                }
            }
        }
        return ans;
    }
};
