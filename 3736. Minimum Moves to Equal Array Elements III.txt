class Solution:
    def minMoves(self, nums: List[int]) -> int:
        mx = nums[0]
        for ele in nums:
            if ele >= mx:
                mx = ele
        
        moves = 0

        for ele in nums:
            x = mx-ele
            moves+=x
        
        return moves
