struct TrieNode {
    bool isEnd = false;
    TrieNode* a = NULL;
    TrieNode* b = NULL;
    TrieNode* c = NULL;
    TrieNode* d = NULL;
    TrieNode* e = NULL;
    TrieNode* f = NULL;
    TrieNode* g = NULL;
    TrieNode* h = NULL;
    TrieNode* i = NULL;
    TrieNode* j = NULL;
    TrieNode* k = NULL;
    TrieNode* l = NULL;
    TrieNode* m = NULL;
    TrieNode* n = NULL;
    TrieNode* o = NULL;
    TrieNode* p = NULL;
    TrieNode* q = NULL;
    TrieNode* r = NULL;
    TrieNode* s = NULL;
    TrieNode* t = NULL;
    TrieNode* u = NULL;
    TrieNode* v = NULL;
    TrieNode* w = NULL;
    TrieNode* x = NULL;
    TrieNode* y = NULL;
    TrieNode* z = NULL;
};

class Trie {
private:
    TrieNode* root;

    // ক্যারেক্টার অনুযায়ী সঠিক ভেরিয়েবলের রেফারেন্স রিটার্ন করার জন্য হেল্পার ফাংশন
    TrieNode*& getChild(TrieNode* curr, char ch) {
        switch (ch) {
            case 'a': return curr->a;
            case 'b': return curr->b;
            case 'c': return curr->c;
            case 'd': return curr->d;
            case 'e': return curr->e;
            case 'f': return curr->f;
            case 'g': return curr->g;
            case 'h': return curr->h;
            case 'i': return curr->i;
            case 'j': return curr->j;
            case 'k': return curr->k;
            case 'l': return curr->l;
            case 'm': return curr->m;
            case 'n': return curr->n;
            case 'o': return curr->o;
            case 'p': return curr->p;
            case 'q': return curr->q;
            case 'r': return curr->r;
            case 's': return curr->s;
            case 't': return curr->t;
            case 'u': return curr->u;
            case 'v': return curr->v;
            case 'w': return curr->w;
            case 'x': return curr->x;
            case 'y': return curr->y;
            case 'z': return curr->z;
            default: throw invalid_argument("Invalid character");
        }
    }

public:
    Trie() { root = new TrieNode(); }

    void insert(string word) {
        TrieNode* curr = root;
        for (char ch : word) {
            TrieNode*& nextNode = getChild(curr, ch);
            if (nextNode == NULL) {
                nextNode = new TrieNode();
            }
            curr = nextNode;
        }
        curr->isEnd = true;
    }

    bool search(string word) {
        TrieNode* curr = root;
        for (char ch : word) {
            TrieNode*& nextNode = getChild(curr, ch);
            if (nextNode == NULL) {
                return false;
            }
            curr = nextNode;
        }
        return curr->isEnd;
    }

    bool startsWith(string prefix) {
        TrieNode* curr = root;
        for (char ch : prefix) {
            TrieNode*& nextNode = getChild(curr, ch);
            if (nextNode == NULL) {
                return false;
            }
            curr = nextNode;
        }
        return true;
    }
};
