class Solution {
public:
    int singleNumber(vector<int>& nums) {
        int len = nums.size();
        int res = 0;
        for(int i=0; i<32; i++)
        {
            int sum = 0;
            for(int j=0; j<len; j++)
            {
                int x = nums[j] >> i & 1;
                sum += x;
            }

            if(sum%3 != 0) res |= (1 << i);

        }

        return res;
    }
};
