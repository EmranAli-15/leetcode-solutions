class Solution {
public:
    int minMoves2(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        int len = nums.size();
        int mid = len/2;
        int total = 0;

        for(int i=0; i<len; i++) {
            int x = abs(nums[i]-nums[mid]);
            total+=x;
        }
        return total;
    }
};
