class Solution {
public:
    void reverseString(vector<char>& s) {
        int r = s.size()-1;
        int l = 0;
        while(l<r){
            char x = s[l];
            s[l] = s[r];
            s[r] = x;
            l++;
            r--;
        }
    }
};
