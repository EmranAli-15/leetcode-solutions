class Solution {
public:
    vector<vector<int>> flipAndInvertImage(vector<vector<int>>& image) {
        int len = image.size();

        for(int i=0; i<len; i++){
            int l=0, r=len-1;
            while(l<r){
                int x = image[i][l];
                image[i][l] = image[i][r];
                image[i][r] = x;
                l++;
                r--;
            }
        }

        for(int i=0; i<len; i++){
            for(int j=0; j<len; j++){
                if(image[i][j]){
                    image[i][j] = 0;
                }else image[i][j] = 1;
            }
        }
        return image;
    }
};
