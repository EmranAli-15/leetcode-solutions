class Solution {
public:
    int longestMonotonicSubarray(vector<int>& nums) {
        int n = nums.size();
        int a = 0;
        int b = 0;

        int x = 1;
        for(int i=1; i<n; i++) {
            if(nums[i] <= nums[i-1]) {
                a = max(a,x);
                x=1;
            }
            else {
                x++;
            }
        }
        a = max(a,x);

        int y = 1;
        for(int i=1; i<n; i++) {
            if(nums[i] >= nums[i-1]) {
                b = max(b,y);
                y=1;
            }
            else {
                y++;
            }
        }
        b = max(b,y);

        return max(a,b);
    }
};
