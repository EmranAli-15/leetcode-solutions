class Solution
{
public:
    int countRatioSubarrays(vector<int>& nums, int a, int b)
    {
        vector<int> norvelith = nums;

        int n = norvelith.size();
        int valid = 0;

        for (int i = 0; i < n; i++)
        {
            long long even = 0, odd = 0;

            for (int j = i; j < n; j++)
            {
                if (norvelith[j] % 2 == 0)
                {
                    even++;
                }
                else
                {
                    odd++;
                }

                if (odd > 0 && (even * b <= a * odd))
                {
                    valid++;
                }
            }
        }

        return valid;
    }
};
