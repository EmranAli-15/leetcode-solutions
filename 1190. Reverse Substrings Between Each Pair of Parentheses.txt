class Solution {
public:
    string reverseParentheses(string s) {
        stack<char>store;

        int len = s.size();
        for(int i=0; i<len; i++)
        {
            char ch = s[i];
            if(ch != ')')store.push(ch);
            else{
                string t = "";
                while(true)
                {
                    char f = store.top();
                    store.pop();
                    if(f != '(')
                    {
                        t+=f;
                    }
                    else{
                        break;
                    }
                }
                for(auto c:t)store.push(c);
            }
        }
        string ans = "";
        while(!store.empty()){
            char a = store.top();
            store.pop();
            ans += a;
        }
        reverse(ans.begin(), ans.end());
        return ans;
    }
};
