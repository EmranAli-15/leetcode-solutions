class Solution {
public:
    bool containsNearbyDuplicate(vector<int>& nums, int k) {
        int l = 0;
        int r = k;
        int n = nums.size();

        if(k>=n)k = n-1;

        unordered_map<int, int> store;
        for(int i=0; i<=k; i++){
            if(store.count(nums[i])) return true;
            else store.insert({nums[i], 1});
        }

        while(r < n-1)
        {
            r++;
            store.erase(nums[l]);
            l++;
            if(store.count(nums[r])) return true;
            else store.insert({nums[r], 1});
        }
        
        return false;
    }
};
