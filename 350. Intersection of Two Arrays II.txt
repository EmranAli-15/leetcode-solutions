class Solution {
public:
    vector<int> intersect(vector<int>& nums1, vector<int>& nums2) {
        int len1 = nums1.size();
        int len2 = nums2.size();
        vector<int> ans;

        for (int i = 0; i < len1; i++) {
            int x = nums1[i];

            for (int j = 0; j < len2; j++) {
                int y = nums2[j];
                if (x == y) {
                    ans.push_back(x);
                    nums2[j] = -1;
                    break;
                }
            }
        }

        return ans;
    }
};
