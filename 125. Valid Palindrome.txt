class Solution {
public:
    bool isPalindrome(string s) {
        string a = "";

        for(auto ch:s){
            if((ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') || (ch >= '0' && ch <= '9')){
                a+=(char)tolower(ch);
            }
        }

        int i=0, j=a.size()-1;
        while(i<j){
            if(a[i] != a[j]) return false;
            i++;
            j--;
        }

        return true;
    }
};
