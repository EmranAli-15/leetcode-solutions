class Solution {
public:
    int smallestIndex(vector<int>& nums) {
        int len = nums.size();

        for(int i=0; i<len; i++) {
            string s = to_string(nums[i]);
            int s_len = s.size();

            int sum = 0;

            for(int j=0; j<s_len; j++) {
                sum += s[j] - '0';
            }

            if(sum == i) return i;
        }

        return -1;
    }
};
