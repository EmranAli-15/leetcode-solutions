/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left),
 * right(right) {}
 * };
 */
class Solution {
public:
    bool fn(TreeNode* node) {
        stack<TreeNode*> ms;
        stack<TreeNode*> ms2;
        ms.push(node);
        ms2.push(node);

        vector<int> a;
        vector<int> b;

        while (!ms.empty()) {
            TreeNode* t = ms.top();
            a.push_back(t->val);
            ms.pop();

            if (t->right && !t->left)
                a.push_back(-300);
            if (!t->right && t->left)
                a.push_back(-400);

            if (t->right)
                ms.push(t->right);
            if (t->left)
                ms.push(t->left);
        }

        //cout << endl;

        while (!ms2.empty()) {
            TreeNode* t = ms2.top();
            b.push_back(t->val);
            ms2.pop();

            if (!t->right && t->left)
                b.push_back(-300);
            if (t->right && !t->left)
                b.push_back(-400);

            if (t->left)
                ms2.push(t->left);
            if (t->right)
                ms2.push(t->right);
        }

        if(a.size() != b.size())
            return false;
        
        int len = a.size();
        
        for (int i=0; i<len; i++)
            if(a[i] != b[i])
                return false;
        
        return true;
    }
    bool isSymmetric(TreeNode* root) {
        bool ans = fn(root);
        return ans;
    }
};
