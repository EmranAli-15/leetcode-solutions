class Solution {
    public boolean uniformArray(int[] nums1) {
        int mn = nums1[0];
        boolean  hasOdd = false;
        boolean  hasEven = false;

        for(int i=0; i<nums1.length; i++){
            if(nums1[i] < mn){
                mn = nums1[i];
            }
            if(nums1[i]%2 == 0) hasEven = true;
            else hasOdd = true;
        }

        if(hasEven && hasOdd){
            if(mn%2 == 0) return false;
            else return true;
        }
        return true;
    }
}
