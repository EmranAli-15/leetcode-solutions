/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left),
 * right(right) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> zigzagLevelOrder(TreeNode* root) {
        vector<vector<int>> ans;
        if (!root)
            return ans;

        deque<TreeNode*> mq;
        mq.push_back(root);
        bool flag = false;

        while (!mq.empty()) {
            int len = mq.size();
            vector<int> temp;

            for (int i = 0; i < len; i++) {
                if (!flag) {
                    TreeNode* node = mq.front();
                    mq.pop_front();
                    
                    temp.push_back(node->val);

                    if (node->left)
                        mq.push_back(node->left);
                    if (node->right)
                        mq.push_back(node->right);
                } else {
                    TreeNode* node = mq.back();
                    mq.pop_back();
                    
                    temp.push_back(node->val);

                    if (node->right)
                        mq.push_front(node->right);
                    if (node->left)
                        mq.push_front(node->left);
                }
            }

            ans.push_back(temp);
            flag = !flag;
        }

        return ans;
    }
};
