class Solution:
    def mul(self, n):
        num = n
        product = 1

        while num > 0:
            digit = num % 10
            product *= digit
            num //= 10
        return product

    def smallestNumber(self, n: int, t: int) -> int:
        for i in range(n, 101):
            res = self.mul(i)
            if res%t == 0:
                return i
