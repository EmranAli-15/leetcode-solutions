class Solution {
public:
    vector<int> resultArray(vector<int>& nums) {
        vector<int>a1;
        vector<int>a2;

        a1.push_back(nums[0]);
        a2.push_back(nums[1]);

        int len = nums.size();

        for(int i=2; i<len; i++)
        {
            int x = nums[i];
            int p = a1[a1.size()-1];
            int q = a2[a2.size()-1];
            if(p > q)a1.push_back(x);
            else a2.push_back(x);
        }

        for(int n:a2)a1.push_back(n);
        return a1;
    }
};
