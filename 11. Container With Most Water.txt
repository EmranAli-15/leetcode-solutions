class Solution {
public:
    int maxArea(vector<int>& height) {
        int i=0, j=height.size()-1;
        int area = 0;

        while(i<j){
            int min_height = min(height[i], height[j]);
            int side = j-i;
            area = max(area, (min_height*side));
            if(height[i] < height[j])i++;
            else j--;
        }

        return area;
    }
};
