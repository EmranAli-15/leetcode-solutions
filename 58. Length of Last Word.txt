class Solution {
public:
    int lengthOfLastWord(string s) {
        string f = "";
        int len = s.size();

        int flag = false;
        for(int i=len-1; i>=0; i--) {
            if(flag && s[i] == ' ')break;
            else if(s[i] != ' '){
                f+=s[i];
                flag = true;
            }
        }

        return f.size();
    }
};
