class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int len = nums.size();
        int l1 = 0, r1 = len - 1, mid1;

        int most_left = -1;

        while (l1 <= r1) {
            mid1 = (l1 + r1) / 2;
            if (nums[mid1] == target) {
                most_left = mid1;
                r1 = mid1 - 1;
            } else if (nums[mid1] < target) {
                l1 = mid1 + 1;
            } else {
                r1 = mid1 - 1;
            }
        }

        if (most_left == -1)
            return {-1, -1};


        int l2 = 0, r2 = len - 1, mid2;
        int most_right = -1;

        while (l2 <= r2) {
            mid2 = (l2 + r2) / 2;
            if (nums[mid2] == target) {
                most_right = mid2;
                l2 = mid2 + 1;
            } else if (nums[mid2] < target) {
                l2 = mid2 + 1;
            } else {
                r2 = mid2 - 1;
            }
        }
        return {most_left, most_right};
    }
};
