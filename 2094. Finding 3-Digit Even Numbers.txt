class Solution {
public:
    vector<int> findEvenNumbers(vector<int>& digits) {
        set<int> store;

        int len = digits.size();

        for(int a=0; a<len; a++){
            if(digits[a] == 0)continue;
            for(int b=0; b<len; b++){
                if(a == b)continue;
                for(int c=0; c<len; c++){
                    if(a==c || b==c)continue;
                    if(digits[c]%2 == 1)continue;
                    int x = 0;
                    x*=10;
                    x+=digits[a];

                    x*=10;
                    x+=digits[b];

                    x*=10;
                    x+=digits[c];

                    store.insert(x);
                }
            }
        }
        vector<int> ans(store.begin(), store.end());
        sort(ans.begin(), ans.end());
        return ans;
    }
};
