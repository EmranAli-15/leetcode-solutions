class Solution {
public:
    int reverseDegree(string s) {
        int len = s.size();
        int sum = 0;

        for(int i=0; i<len; i++) {
            int x = s[i];
            sum += ((26-(x-97))*(i+1));
        }
        return sum;
    }
};
