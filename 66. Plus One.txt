class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        vector<int> ans;
        int len = digits.size();

        int carry = 1;
        int rem = 0;
        for(int i=len-1; i>=0; i--)
        {
            int x = digits[i];
            x = x + carry;
            carry = x/10;
            ans.push_back(x%10);
        }
        if(carry)ans.push_back(carry);
        reverse(ans.begin(), ans.end());
        return ans;
    }
};
