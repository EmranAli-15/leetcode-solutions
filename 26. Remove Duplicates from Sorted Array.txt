class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int len = nums.size();
        if(len == 1) return 1;

        for(int i=1; i<len; i++)
        {
            if(nums[i-1] == nums[i])
            {
                nums[i] = 200;
                swap(nums[i], nums[i-1]);
            }
        };   

        int left = 0, right = 0;

        for(int i=0; i<len; i++) 
        {
            if(nums[i] == 200)
            {
                left = i;
                break;
            }
        }

        for(int i=0; i<len; i++)
        {
            if(nums[i] == 200)
            {
                right = i+1;
                break;
            }
        }

        while(right < len)
        {
            if(nums[left] == 200 && nums[right] != 200)
            {
                int temp = nums[right];
                nums[right] = nums[left];
                nums[left] = temp;
                left++;
            }
            else if(nums[left] != 200) left++;
            
            right++;
        }
        return left;
    }
};
