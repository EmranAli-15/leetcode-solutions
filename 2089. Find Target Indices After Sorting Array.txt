class Solution {
public:
    vector<int> targetIndices(vector<int>& nums, int target) {
        sort(nums.begin(), nums.end());
        vector<int> ans;
        int len = nums.size();
        for(int i=0; i<len; i++) {
            if(nums[i] == target) ans.push_back(i);
        };

        return ans;
    }
};
