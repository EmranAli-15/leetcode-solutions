class Solution:
    def strStr(self, haystack: str, needle: str) -> int:
        lps = [0] * len(needle)
        i = 1
        l = 0

        while i < len(needle):
            if needle[i] == needle[l]:
                l += 1
                lps[i] = l
                i += 1
            else:
                if l != 0:
                    l = lps[l - 1]
                else:
                    i += 1

        k = 0
        h = 0
        ans = -1

        while h < len(haystack):
            if haystack[h] == needle[k]:
                h += 1
                k += 1
            else:
                if k != 0:
                    k = lps[k - 1]
                else:
                    h += 1

            if k == len(needle):
                ans = h-k
                break
        return ans
