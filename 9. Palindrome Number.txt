class Solution {
public:
    bool isPalindrome(int x) {
        if(x < 0 || (x%10 == 0 && x != 0)) return false;

        int revert = 0;
        int rem;
        while(x > revert)
        {
            rem = x%10;
            revert = revert*10 + rem;
            x = x/10;
        }

        if(x == revert || x == revert/10) return true;
        return false;
    }
};
