class Solution {
public:
    string reverseWords(string s) {
        int len = s.size();

        vector<string> str;

        string temp = "";
        for(int i=0; i<len; i++)
        {
            if(s[i] == ' ' && temp.size() > 0)
            {
                str.push_back(temp);
                temp = "";
            }
            else if(s[i] == ' ' && temp.size()==0)continue;
            else temp+=s[i];
        }

        if(temp.size() !=0)str.push_back(temp);

        string ans = "";
        for(int i=str.size()-1; i>=0; i--)
        {
            ans+=str[i];
            if (i>0)
                ans+=" ";
        }
        return ans;
    }
};
