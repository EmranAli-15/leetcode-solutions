class Solution
{
public:
    unordered_map<char, int> letters;
    int maxLen = 0;

    int lengthOfLongestSubstring(string s)
    {
        int len = s.size();
        if(len == 0) return maxLen;

        for(int i=0; i<len; i++)
        {
            char a = s[i];
            if (letters.find(a) != letters.end())
            {
                int localLen = letters.size();
                if(localLen > maxLen) maxLen = localLen;
                letters.clear();

                int rev = i-1;
                while(rev >= 0)
                {
                    char n = s[rev];
                    if(n != a)
                    {
                        letters.insert({n, 1});
                        rev--;
                    }
                    else break;
                }

                letters.insert({a, 1});
            }
            else
            {
                letters.insert({a, 1});
            }
        }

        int localLen = letters.size();
        if(localLen > maxLen) maxLen = localLen;

        return maxLen;
    }
};
