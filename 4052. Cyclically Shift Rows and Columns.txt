class Solution:
    def cyclicShift(self, n: int, grid: list[list[int]], rowShift: list[int], colShift: list[int]) -> list[list[int]]:
        myGrid = [[0] * n for i in range(n)]
        for i in range(n):
            k = rowShift[i]%n
            for j in range(n):
                myGrid[i][(j-k+n)%n] = grid[i][j]

        ans = [[0] * n for i in range(n)]
        for k in range(n):
            p = colShift[k]%n
            for z in range(n):
                ans[(z-p+n)%n][k] = myGrid[z][k]

        return ans
