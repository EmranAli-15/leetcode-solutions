class Solution {
public:
    int numOfSubarrays(vector<int>& arr, int k, int threshold) {
        int ans = 0;
        int n = arr.size();

        int j=k-1;
        
        int sum = 0;
        for(int i=0; i<k; i++)sum+=arr[i];
        if(sum/k >= threshold) ans++;
        j++;

        while(j<n)
        {
            sum += arr[j];
            sum -= arr[j-k];
            if(sum/k >= threshold) ans++;
            j++;
        }
        return ans;
    }
};
