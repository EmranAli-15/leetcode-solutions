class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        priority_queue<int, vector<int>, greater<int>> min_pq;

        int len = nums.size();
        for(int i=0; i<len; i++)
        {
            if(min_pq.size() == k){
                int x = min_pq.top();
                if(x<nums[i]){
                    min_pq.pop();
                    min_pq.push(nums[i]);
                }
            }
            else min_pq.push(nums[i]);
        }

        return min_pq.top();
    }
};
