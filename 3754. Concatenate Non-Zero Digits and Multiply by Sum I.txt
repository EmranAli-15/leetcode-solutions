class Solution {
public:
    long long sumAndMultiply(int n) {
        string a = to_string(n);
        long long sum = 0;
        long long f = 0;
        for(auto x:a) {
            long long c = x - '0';
            if(c){
                sum+=c;
                f*=10;
                f+=c;
            }
        }
        long long ans = sum*f;
        return ans;
    }
};
