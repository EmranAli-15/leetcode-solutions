        : val(_val), left(_left), right(_right), next(_next) {}

class Solution {
public:
    Node* connect(Node* root) {
        if(root == NULL) return NULL;
        vector<Node*> nodes;
        queue<Node*> mq;
        mq.push(root);
        nodes.push_back(root);
        while(!mq.empty()) {
            Node* x = mq.front();
            if(x->left) {
                mq.push(x->left);
                nodes.push_back(x->left);
            }
            if(x->right) {
                mq.push(x->right);
                nodes.push_back(x->right);
            }
            mq.pop();
        }
        
        int twise = 1;
        int passed = 0;
        int len = nodes.size();

        for(int i=0; i<len; i++) {
            if((passed + twise) == i+1) {
                //cout << "next null -> " << passed + twise << endl;
                nodes[i]->next = NULL;
                passed =  i+1;
                twise*=2;
            }
            else {
                nodes[i]->next = nodes[i+1];
            }
        }

        return root;
    }
};
