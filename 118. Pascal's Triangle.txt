class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        if(numRows == 1) return {{1}};
        if(numRows == 2) return {{1}, {1,1}};

        vector<vector<int>>ans = {{1}, {1,1}};

        for(int i=1; i<numRows-1; i++) {
            vector<int> t = {1};
            int len = ans[i].size();
            for(int j=0; j<len-1; j++) {
                t.push_back(ans[i][j] + ans[i][j+1]);
            }
            t.push_back(1);
            ans.push_back(t);
        }

        return ans;
    }
};
