class Solution {
public:
    int sumCounts(vector<int>& nums) {
        int len = nums.size();
        int ans = 0;

        for(int i=0; i<len; i++) {
            for(int j=i; j<len; j++) {
                set<int> ele;
                for(int k=i; k<=j; k++) {
                    ele.insert(nums[k]);
                }
                int ele_len = ele.size();
                ans+=(ele_len * ele_len);
            }
        }
        return ans;
    }
};
