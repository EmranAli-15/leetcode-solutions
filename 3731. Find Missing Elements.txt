class Solution {
public:
    vector<int> findMissingElements(vector<int>& nums) {
        int n = nums.size();
        sort(nums.begin(), nums.end());
        vector<int> ans;
        for(int i=1; i<n; i++)
        {
            int x = nums[i-1];
            int y = nums[i];
            int z = y-x;

            if(z>1)
            {
                for(int i=1; i<z; i++)
                {
                    ans.push_back(x+i);
                }
            }
        }
        return ans;
    }
};
